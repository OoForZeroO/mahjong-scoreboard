# 服务器脚本模板

这些脚本需要上传到服务器的 `/opt/yaohufox/scripts/` 目录。

## 1. 数据库备份脚本

文件：`/opt/yaohufox/scripts/backup-db.sh`

```bash
#!/bin/bash

BACKUP_DIR="/opt/yaohufox/data/backup"
DATE=$(date +%Y%m%d_%H%M%S)

# 创建备份目录
mkdir -p "$BACKUP_DIR"

# 备份生产数据库
echo "开始备份生产数据库..."
docker exec yaohufox-postgres-prod pg_dump -U mahjong_user mahjong_score_system | gzip > "$BACKUP_DIR/production_$DATE.sql.gz"

if [ $? -eq 0 ]; then
    echo "生产数据库备份成功：production_$DATE.sql.gz"
else
    echo "生产数据库备份失败！"
    exit 1
fi

# 备份测试数据库
echo "开始备份测试数据库..."
docker exec yaohufox-postgres-test pg_dump -U mahjong_user_test mahjong_score_system_test | gzip > "$BACKUP_DIR/testing_$DATE.sql.gz"

if [ $? -eq 0 ]; then
    echo "测试数据库备份成功：testing_$DATE.sql.gz"
else
    echo "测试数据库备份失败！"
    exit 1
fi

# 删除 7 天前的备份
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +7 -delete

echo "备份完成：$DATE"
echo "备份文件保存在：$BACKUP_DIR"
```

**使用方法：**
```bash
chmod +x /opt/yaohufox/scripts/backup-db.sh

# 手动执行
/opt/yaohufox/scripts/backup-db.sh

# 设置定时任务（每天凌晨 2 点）
crontab -e
# 添加：0 2 * * * /opt/yaohufox/scripts/backup-db.sh
```

## 2. 日志清理脚本

文件：`/opt/yaohufox/scripts/cleanup-logs.sh`

```bash
#!/bin/bash

echo "开始清理日志..."

# 清理 30 天前的应用日志
find /opt/yaohufox/logs/app -name "*.log" -mtime +30 -delete
echo "已清理 30 天前的应用日志"

# 清理 30 天前的 Nginx 日志
find /opt/yaohufox/logs/nginx -name "*.log" -mtime +30 -delete
echo "已清理 30 天前的 Nginx 日志"

# 清理 Docker 日志（保留最近 7 天）
docker system prune -f --volumes
echo "已清理 Docker 日志"

echo "日志清理完成"
```

**使用方法：**
```bash
chmod +x /opt/yaohufox/scripts/cleanup-logs.sh

# 手动执行
/opt/yaohufox/scripts/cleanup-logs.sh

# 设置定时任务（每周日凌晨 3 点）
crontab -e
# 添加：0 3 * * 0 /opt/yaohufox/scripts/cleanup-logs.sh
```

## 3. 服务重启脚本

文件：`/opt/yaohufox/scripts/restart-production.sh`

```bash
#!/bin/bash

echo "重启生产环境服务..."

cd /opt/yaohufox/production

# 重启应用
echo "重启应用容器..."
docker-compose restart mahjong-app-prod

# 等待应用启动
echo "等待应用启动..."
sleep 10

# 检查健康状态
echo "检查应用健康状态..."
curl -f http://localhost:8081/actuator/health

if [ $? -eq 0 ]; then
    echo "生产环境重启成功！"
else
    echo "生产环境重启失败，请检查日志！"
    docker-compose logs --tail=50 mahjong-app-prod
    exit 1
fi
```

## 4. 服务状态检查脚本

文件：`/opt/yaohufox/scripts/check-status.sh`

```bash
#!/bin/bash

echo "========== 服务状态检查 =========="
echo ""

echo "--- Docker 容器状态 ---"
docker ps --filter "name=yaohufox" --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"
echo ""

echo "--- 生产环境健康检查 ---"
curl -s http://localhost:8081/actuator/health | python3 -m json.tool 2>/dev/null || echo "生产环境不可用"
echo ""

echo "--- 测试环境健康检查 ---"
curl -s http://localhost:8082/actuator/health | python3 -m json.tool 2>/dev/null || echo "测试环境不可用"
echo ""

echo "--- 磁盘使用情况 ---"
df -h /opt/yaohufox
echo ""

echo "--- 最近的应用日志（生产环境） ---"
tail -n 20 /opt/yaohufox/logs/app/production/application.log 2>/dev/null || echo "日志文件不存在"
echo ""

echo "========== 检查完成 =========="
```

**使用方法：**
```bash
chmod +x /opt/yaohufox/scripts/check-status.sh

# 执行检查
/opt/yaohufox/scripts/check-status.sh
```

## 5. 数据库恢复脚本

文件：`/opt/yaohufox/scripts/restore-db.sh`

```bash
#!/bin/bash

if [ $# -ne 2 ]; then
    echo "使用方法: $0 <环境> <备份文件>"
    echo "环境: production 或 testing"
    echo "示例: $0 production /opt/yaohufox/data/backup/production_20241120_020000.sql.gz"
    exit 1
fi

ENV=$1
BACKUP_FILE=$2

if [ ! -f "$BACKUP_FILE" ]; then
    echo "错误：备份文件不存在：$BACKUP_FILE"
    exit 1
fi

if [ "$ENV" = "production" ]; then
    CONTAINER="yaohufox-postgres-prod"
    DB_NAME="mahjong_score_system"
    DB_USER="mahjong_user"
elif [ "$ENV" = "testing" ]; then
    CONTAINER="yaohufox-postgres-test"
    DB_NAME="mahjong_score_system_test"
    DB_USER="mahjong_user_test"
else
    echo "错误：环境必须是 production 或 testing"
    exit 1
fi

echo "警告：此操作将覆盖 $ENV 环境的数据库！"
read -p "确认继续？(yes/no): " confirm

if [ "$confirm" != "yes" ]; then
    echo "操作已取消"
    exit 0
fi

echo "开始恢复 $ENV 环境数据库..."

# 解压并恢复
gunzip -c "$BACKUP_FILE" | docker exec -i "$CONTAINER" psql -U "$DB_USER" -d "$DB_NAME"

if [ $? -eq 0 ]; then
    echo "数据库恢复成功！"
else
    echo "数据库恢复失败！"
    exit 1
fi
```

**使用方法：**
```bash
chmod +x /opt/yaohufox/scripts/restore-db.sh

# 恢复生产环境
/opt/yaohufox/scripts/restore-db.sh production /opt/yaohufox/data/backup/production_20241120_020000.sql.gz

# 恢复测试环境
/opt/yaohufox/scripts/restore-db.sh testing /opt/yaohufox/data/backup/testing_20241120_020000.sql.gz
```

## 6. 一键部署脚本（用于手动部署）

文件：`/opt/yaohufox/scripts/deploy.sh`

```bash
#!/bin/bash

ENV=$1

if [ -z "$ENV" ]; then
    echo "使用方法: $0 <环境>"
    echo "环境: production 或 testing"
    exit 1
fi

if [ "$ENV" != "production" ] && [ "$ENV" != "testing" ]; then
    echo "错误：环境必须是 production 或 testing"
    exit 1
fi

echo "开始部署 $ENV 环境..."

cd "/opt/yaohufox/$ENV"

# 停止旧容器
echo "停止旧容器..."
docker-compose down

# 拉取最新镜像（如果需要）
echo "拉取最新镜像..."
docker-compose pull || true

# 启动新容器
echo "启动新容器..."
docker-compose --env-file .env up -d

# 等待服务启动
echo "等待服务启动..."
sleep 15

# 检查健康状态
if [ "$ENV" = "production" ]; then
    HEALTH_URL="http://localhost:8081/actuator/health"
else
    HEALTH_URL="http://localhost:8082/actuator/health"
fi

echo "检查服务健康状态..."
for i in {1..10}; do
    if curl -f "$HEALTH_URL" > /dev/null 2>&1; then
        echo "$ENV 环境部署成功！"
        exit 0
    fi
    echo "等待服务启动... ($i/10)"
    sleep 3
done

echo "$ENV 环境部署失败，请检查日志！"
docker-compose logs --tail=50
exit 1
```

**使用方法：**
```bash
chmod +x /opt/yaohufox/scripts/deploy.sh

# 部署生产环境
/opt/yaohufox/scripts/deploy.sh production

# 部署测试环境
/opt/yaohufox/scripts/deploy.sh testing
```
